CRO fitting with CESM2#

This tutorial illustrates how to use the mCRO to estimate RO parameters from CESM2

This notebook was contributed by Soong-Ki Kim.

Fitting RO to the CESM2 simulation#

Load ENSO time series from CESM2#

We begin by loading monthly Niño3.4 SST anomalies and equatorial Pacific Warm Water Volume (WWV; stored as Hm in the CRO CMIP6 dataset) from the CESM2 historical simulation.

The Python version of CRO uses pyCRO.ROdata_load(name=”CMIP6”) and then selects model=”CESM2”. The MATLAB version does not include an equivalent convenience loader, so the NetCDF file is loaded manually below.

[2]:

% CESM2 is the 8th CMIP6 model in the CRO model ordering.
model_index=8;


fpath=pwd+"/Data/"+"CROdata_timeseries_cmip6.nc"; % replace the filename/path if needed


Nino34_all=ncread(fpath,'Nino34');
Hm_all=ncread(fpath,'Hm');


% The CRO CMIP6 file is expected as [model x time].
% Extract CESM2 and convert to column vectors for RO_fitting.
T_cesm2=squeeze(Nino34_all(:,model_index));
h_cesm2=squeeze(Hm_all(:,model_index));
T_cesm2=T_cesm2(:);
h_cesm2=h_cesm2(:);


% Historical time axis used in this tutorial (monthly, beginning January 1900).
time_cesm2=(1900+(0:numel(T_cesm2)-1)'/12);


ncdisp(fpath)

Local helper functions#

These helper functions replace convenience functions used by the Python notebook.

[14]:

function S=seasonal_statistic(X,stat_name)
% Return [12 x ensemble] seasonal statistics, assuming January is the first sample.


if isvector(X)
    X=X(:);
end


nMember=size(X,2);
S=nan(12,nMember);
for m=1:12
    idx=m:12:size(X,1);
    for im=1:nMember
        x=X(idx,im);
        switch lower(stat_name)
            case "std"
                S(m,im)=std(x,0,'omitnan');
            otherwise
                error("Unknown statistic: %s",stat_name)
        end
    end
end
end


function h=plot_mean_spread(x,Y,label_text,line_color)
% Plot ensemble mean and 10th-90th percentile spread.


if isvector(Y)
    Y=Y(:);
end


mu=mean(Y,2,'omitnan');
lo=prctile(Y,10,2);
hi=prctile(Y,90,2);


h=plot(x,mu,'Color',line_color,'LineWidth',1.8,'DisplayName',label_text);
hold on
fill([x(:);flipud(x(:))],[lo(:);flipud(hi(:))],line_color, ...
    'FaceAlpha',0.20,'EdgeColor','none','HandleVisibility','off');
uistack(h,'top')
end


function plot_RO_par_comparison(par_list,label_list)
% MATLAB replacement for pyCRO.plot_RO_par used in this tutorial.
% Reconstruct monthly deterministic parameters from scalar/seasonal CRO coefficients.


months=1:12;
month_label={'J','F','M','A','M','J','J','A','S','O','N','D'};


% Panel definitions: BWJ, R, epsilon, F1, F2, b_T, c_T, d_T, b_h
panel_par={0,1,4,2,3,5,6,7,8};
panel_title={'BJ','R','\epsilon','F_1','F_2','b_T','c_T','d_T','b_h'};
panel_unit={'month^{-1}','month^{-1}','month^{-1}', ...
    'K m^{-1} month^{-1}','m K^{-1} month^{-1}', ...
    'K^{-1} month^{-1}','K^{-2} month^{-1}', ...
    'm^{-1} month^{-1}','K^{-2} m month^{-1}'};


figure;
tiledlayout(3,3,'TileSpacing','compact','Padding','compact');


for ipanel=1:9
    nexttile
    hold on

    for imodel=1:numel(par_list)
        par=par_list{imodel};

        if panel_par{ipanel}==0

            R_month=arrayfun(@(m) monthly_parameter_value(par{1},m),months);
            epsilon_month=arrayfun(@(m) monthly_parameter_value(par{4},m),months);

            y=0.5*R_month;

        else

            k=panel_par{ipanel};
            y=arrayfun(@(m) monthly_parameter_value(par{k},m),months);

        end

        plot(months,y,'-o','LineWidth',1.3,'MarkerSize',3.5, ...
            'DisplayName',label_list{imodel});
    end


    xlim([0.5 12.5])
    xticks(1:12)
    xticklabels(month_label)
    ylabel(sprintf('%s (%s)',panel_title{ipanel},panel_unit{ipanel}), ...
        'Interpreter','tex')
    grid on


    if ipanel==9
        legend('Location','best','FontSize',8)
    end
end
end


function y=monthly_parameter_value(v,month_index)
% Evaluate a CRO parameter at one calendar month.
% []/NaN parameters are plotted as zero, matching the Python tutorial display.


if isempty(v) || all(~isfinite(v))
    y=0;
    return
end


v=v(:).';
t=(month_index-1);
w=2*pi/12;


switch numel(v)
    case 1
        y=v(1);
    case 3
        y=v(1)+v(2)*sin(w*t+v(3));
    case 5
        y=v(1)+v(2)*sin(w*t+v(3))+v(4)*sin(2*w*t+v(5));
    otherwise
        % Time-varying user input: use the corresponding month cyclically.
        idx=mod(month_index-1,numel(v))+1;
        y=v(idx);
end
end
